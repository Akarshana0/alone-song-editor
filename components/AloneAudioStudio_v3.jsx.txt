function useCDNLibs() {
  const [state, setState] = useState({ audio: false, ffmpeg: false });

  useEffect(() => {
    const load = (src) =>
      new Promise((res, rej) => {
        if (document.querySelector(`script[src="${src}"]`)) { res(); return; }
        const s = document.createElement("script");
        s.src = src; s.async = true;
        s.onload = res; s.onerror = rej;
        document.head.appendChild(s);
      });

    // Audio libs (must be ready before tracks can play)
    Promise.all([
      load("https://unpkg.com/wavesurfer.js@7/dist/wavesurfer.min.js"),
      load("https://cdnjs.cloudflare.com/ajax/libs/tone/14.8.49/Tone.js"),
    ])
      .then(() => setState((s) => ({ ...s, audio: true })))
      .catch((e) => console.error("Audio CDN failed:", e));

    // FFmpeg libs (lazy — loaded in background, ready before export)
    Promise.all([
      load("https://unpkg.com/@ffmpeg/ffmpeg@0.12.10/dist/umd/ffmpeg.js"),
      load("https://unpkg.com/@ffmpeg/util@0.12.1/dist/umd/index.js"),
    ])
      .then(() => setState((s) => ({ ...s, ffmpeg: true })))
      .catch((e) => console.warn("FFmpeg CDN failed (export will be unavailable):", e));
  }, []);

  return state;
}

let _clipSeed = 1;

/**
 * Split the clip at `time` (seconds within timeline).
 * Returns a new clips array with the hit clip replaced by two halves.
 */
function cutClipAt(clips, time) {
  const idx = clips.findIndex(
    (c) => c.active && time > c.start + 0.05 && time < c.end - 0.05
  );
  if (idx === -1) return clips; // too close to edge or no clip found

  const clip = clips[idx];
  const leftClip  = { ...clip, end: time };
  const rightClip = {
    id:          newClipId(),
    start:       time,
    end:         clip.end,
    audioOffset: clip.audioOffset + (time - clip.start),
    active:      true,
  };

  return [
    ...clips.slice(0, idx),
    leftClip,
    rightClip,
    ...clips.slice(idx + 1),
  ];
}

// Shape per trackId: { player, eq10, vol, meter }
const NODES = new Map();

const EQ_KEYS = ["32","64","125","250","500","1000","2000","4000","8000","16000"];

function buildChain(trackId) {
  if (!window.Tone) return null;
  teardown(trackId);
  const T = window.Tone;
  const eq10   = new T.EQ10({ 32:0,64:0,125:0,250:0,500:0,"1000":0,"2000":0,"4000":0,"8000":0,"16000":0 });
  const vol    = new T.Volume(0);
  const meter  = new T.Meter({ normalRange: true, smoothing: 0.8 });
  eq10.chain(vol, meter, T.getDestination());
  const player = new T.Player({ loop: false }).connect(eq10);
  NODES.set(trackId, { player, eq10, vol, meter });
  return NODES.get(trackId);
}

function teardown(trackId) {
  const n = NODES.get(trackId);
  if (!n) return;
  try { n.player.stop(); n.player.dispose(); n.eq10.dispose(); n.vol.dispose(); n.meter.dispose(); } catch {}
  NODES.delete(trackId);
}

function getNodes(id) { return NODES.get(id); }

function CyberButton({ children, active, color = C.cyan, onClick, small = false,
                       danger = false, disabled = false, className = "" }) {
  const col = danger ? C.pink : color;
  return (
    <motion.button
      onClick={disabled ? undefined : onClick}
      whileHover={disabled ? {} : { scale: 1.04 }}
      whileTap={disabled ? {} : { scale: 0.95 }}
      className={`relative overflow-hidden font-mono tracking-widest uppercase select-none ${className}`}
      style={{
        padding:    small ? "4px 10px" : "7px 18px",
        fontSize:   small ? "9px" : "10px",
        border:     `1px solid ${active && !disabled ? col : C.border}`,
        background: active && !disabled ? `rgba(${rgb(col)},0.12)` : "transparent",

function VUMeter({ trackId, active, color = C.cyan }) {
  const [lv, setLv] = useState(0);
  useEffect(() => {
    if (!active) { setLv(0); return; }
    const id = setInterval(() => {
      const n = getNodes(trackId);
      const v = n?.meter ? n.meter.getValue() : Math.random() * 0.3;
      setLv(typeof v === "number" ? v : 0);
    }, 60);
    return () => clearInterval(id);
  }, [active, trackId]);

  return (
    <div style={{ display:"flex",gap:2,alignItems:"flex-end",height:22 }}>
      {Array.from({ length:12 }, (_, i) => {
        const thresh = (i + 1) / 12;
        const lit    = lv >= thresh;
        const col    = i > 10 ? C.pink : i > 8 ? C.yellow : color;
        return (
          <div key={i} style={{ width:3,minHeight:2,
            height: lit ? `${thresh*100}%` : "12%",
            background: lit ? col : C.border,
            transition:"height 0.06s ease", opacity: lit ? 1 : 0.3 }}/>
        );
      })}
    </div>
  );
}

const EQ_LABELS = ["32","64","125","250","500","1K","2K","4K","8K","16K"];

function EQCurve({ gains }) {
  const W = 420, H = 52, mid = H / 2;
  const pts = gains.map((g, i) => ({ x: (i / (gains.length - 1)) * W, y: mid - (g / 12) * (H * 0.42) }));
  const path = `M ${pts.map((p) => `${p.x},${p.y}`).join(" L ")}`;
  const fill = `${path} L ${W},${mid} L 0,${mid} Z`;
  return (
    <svg width={W} height={H} style={{ display:"block",overflow:"visible",maxWidth:"100%" }}>
      <line x1={0} y1={mid} x2={W} y2={mid} stroke={C.border} strokeWidth={1}/>
      <defs>
        <linearGradient id="eqgrad2" x1="0" y1="0" x2="0" y2="1">
          <stop offset="0%" stopColor={C.cyan} stopOpacity={0.18}/>
          <stop offset="100%" stopColor={C.cyan} stopOpacity={0}/>
        </linearGradient>
      </defs>
      <path d={fill} fill="url(#eqgrad2)"/>
      <path d={path} fill="none" stroke={C.cyan} strokeWidth={1.5} opacity={0.85}/>
      {pts.map((p, i) => <circle key={i} cx={p.x} cy={p.y} r={3.5} fill={C.cyan} opacity={0.75}/>)}
    </svg>
  );
}

              </motion.div>
            )}
          </div>
        ))}
      </div>
    </div>
  );
}

// ═══════════════════════════════════════════════════════════════════════════════
//  TIMELINE RULER
// ═══════════════════════════════════════════════════════════════════════════════
function TimelineRuler({ maxDuration = 120 }) {
  const step  = maxDuration > 180 ? 30 : maxDuration > 90 ? 15 : 10;
  const marks = Array.from({ length: Math.ceil(maxDuration / step) + 1 }, (_, i) => i * step);
  return (
    <div style={{ height:24,background:C.bgPanel,borderBottom:`1px solid ${C.border}`,
                  display:"flex",alignItems:"flex-end",paddingBottom:2,paddingLeft:180,position:"relative",overflow:"hidden" }}>
      {marks.map((t) => (
        <div key={t} style={{ position:"absolute",left:`calc(180px + ${(t/maxDuration)*100}%)`,
                              display:"flex",flexDirection:"column",alignItems:"center" }}>
          <span style={{ fontFamily:"monospace",fontSize:7,color:C.textDim }}>
            {`${Math.floor(t/60)}:${String(t%60).padStart(2,"0")}`}
          </span>
          <div style={{ width:1,height:6,background:C.border }}/>
        </div>
      ))}
    </div>
  );
}

// ═══════════════════════════════════════════════════════════════════════════════
//  CLIP CANVAS  — [Step 4] overlays the WaveSurfer waveform
//  Renders clip boundaries, cut markers, selection highlight, deleted hatching.
//  Forwards pointer events so cut/select tool clicks are handled here.
// ═══════════════════════════════════════════════════════════════════════════════
function ClipCanvas({ clips, duration, color, tool, onCut, onSelectClip, selectedClipId }) {
  const canvasRef = useRef(null);

  // ── Draw ────────────────────────────────────────────────────────────────────
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const container = canvas.parentElement;
    const W = container?.clientWidth  || 600;
    const H = container?.clientHeight || 52;

    canvas.width  = W;
    canvas.height = H;

    const ctx = canvas.getContext("2d");
    ctx.clearRect(0, 0, W, H);

    if (!duration || clips.length === 0) return;

    const px = (t) => (t / duration) * W;

    clips.forEach((clip, idx) => {
      const x1 = px(clip.start);
      const x2 = px(clip.end);
      const w  = Math.max(0, x2 - x1);

      if (!clip.active) {
        // ── Deleted clip: dark hatch ──────────────────────────────────────
        ctx.fillStyle = "rgba(4,4,14,0.78)";
        ctx.fillRect(x1, 0, w, H);
        ctx.save();
        ctx.beginPath();
        ctx.rect(x1, 0, w, H);
        ctx.clip();
        ctx.strokeStyle = "rgba(80,90,115,0.35)";
        ctx.lineWidth   = 1;
        for (let d = -H; d < w + H; d += 9) {
          ctx.beginPath();
          ctx.moveTo(x1 + d, 0);
          ctx.lineTo(x1 + d + H, H);
          ctx.stroke();
        }
        ctx.restore();
        // Label
        if (w > 40) {
          ctx.fillStyle = "rgba(80,90,115,0.7)";
          ctx.font = "9px monospace";
          ctx.fillText("✕ REMOVED", x1 + 6, H / 2 + 3);
        }
      } else if (clip.id === selectedClipId) {
        // ── Selected clip: glow + outline ─────────────────────────────────
        ctx.fillStyle = `rgba(0,243,255,0.09)`;
        ctx.fillRect(x1, 0, w, H);
        ctx.strokeStyle = `rgba(0,243,255,0.85)`;
        ctx.lineWidth   = 1.5;
        ctx.setLineDash([]);
        ctx.strokeRect(x1 + 1, 1, w - 2, H - 2);
        // Label
        if (w > 50) {
          ctx.fillStyle = `rgba(0,243,255,0.55)`;
          ctx.font      = "8px monospace";
          ctx.fillText("SELECTED · DEL TO REMOVE", x1 + 6, H - 5);
        }
      }

      // ── Cut marker at each non-first clip boundary ────────────────────────
      if (idx > 0 && clip.active) {
        const cx = x1;

        // Dashed vertical rule
        ctx.save();
        ctx.strokeStyle = `rgba(${rgb(color)},0.9)`;
        ctx.lineWidth   = 1.5;
        ctx.setLineDash([3, 3]);
        ctx.beginPath();
        ctx.moveTo(cx, 0);
        ctx.lineTo(cx, H);
        ctx.stroke();
        ctx.setLineDash([]);
        ctx.restore();

        // Triangle cap at top
        ctx.fillStyle = `rgba(${rgb(color)},1)`;
        ctx.beginPath();
        ctx.moveTo(cx - 5, 0);
        ctx.lineTo(cx + 5, 0);
        ctx.lineTo(cx, 9);
        ctx.closePath();
        ctx.fill();

        // Scissors icon area (subtle)
        ctx.fillStyle = `rgba(${rgb(color)},0.25)`;
        ctx.fillRect(cx - 1, 0, 2, H);
      }
    });

    // ── Cut-tool cursor hint: show ✂ label when tool is "cut" and track has audio
    if (tool === "cut" && duration > 0) {
      ctx.fillStyle = `rgba(${rgb(color)},0.25)`;
      ctx.font      = "8px monospace";
      ctx.fillText("✂  CLICK TO CUT", 6, H - 5);
    }

  }, [clips, duration, color, selectedClipId, tool]);

  // ── Re-draw when container resizes ─────────────────────────────────────────
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ro = new ResizeObserver(() => {
      const W = canvas.parentElement?.clientWidth  || 600;
      const H = canvas.parentElement?.clientHeight || 52;
      canvas.width  = W;
      canvas.height = H;
      // Re-trigger the draw effect by dispatching a synthetic update
      canvas.dispatchEvent(new Event("resize-redraw"));
    });
    if (canvas.parentElement) ro.observe(canvas.parentElement);
    return () => ro.disconnect();
  }, []);

  // ── Click handler ──────────────────────────────────────────────────────────
  const handleClick = useCallback((e) => {
    if (!duration) return;
    const rect = e.currentTarget.getBoundingClientRect();
    const x    = e.clientX - rect.left;
    const time = (x / rect.width) * duration;

    if (tool === "cut") {
      onCut(time);
    } else if (tool === "select") {
      const hit = clips.find((c) => time >= c.start && time < c.end);
      onSelectClip(hit ? hit.id : null);
    }
  }, [duration, tool, clips, onCut, onSelectClip]);

  return (
    <canvas
      ref={canvasRef}
      onClick={handleClick}
      style={{
        position: "absolute",
        inset:    0,
        width:    "100%",
        height:   "100%",
        zIndex:   3,
        cursor:   tool === "cut" ? "col-resize" : tool === "select" ? "crosshair" : "default",
        // Only capture clicks when cut or select tool is active
        pointerEvents: (tool === "cut" || tool === "select") && duration ? "auto" : "none",
        background: "transparent",
      }}
    />
  );
}

// ═══════════════════════════════════════════════════════════════════════════════
//  EXPORT MODAL  — [Step 5] FFmpeg.wasm client-side mix + download
// ═══════════════════════════════════════════════════════════════════════════════
function ExportModal({ tracks, libsReady, onClose }) {
  const [stage,    setStage]   = useState("idle");
  // idle | loading-core | processing | done | error
  const [progress, setProgress] = useState(0);
  const [statusMsg,setStatus]   = useState("");
  const [format,   setFormat]   = useState("wav");
  const [errorMsg, setError]    = useState("");
  const ffmpegRef = useRef(null);

  const exportableTracks = tracks.filter((t) => !t.muted && t.loaded && BLOB_URLS.has(t.id));
  const canExport = exportableTracks.length > 0 && libsReady.ffmpeg;
  const isBusy    = stage === "loading-core" || stage === "processing";

  const runExport = async () => {
    if (!canExport) return;
    setError("");

    try {
      // ── 1. Load FFmpeg WASM core ────────────────────────────────────────
      setStage("loading-core");
      setProgress(0);
      setStatus("LOADING FFMPEG WASM CORE (~10 MB)...");

      const { FFmpeg }     = window.FFmpegWASM;
      const { fetchFile }  = window.FFmpegUtil;

      const ffmpeg = new FFmpeg();
      ffmpegRef.current = ffmpeg;

      ffmpeg.on("log", ({ message }) => {
        if (message.includes("time=") || message.includes("size=")) {
          setStatus(message.slice(0, 72));
        }
      });

      // Try direct URL first; fall back to toBlobURL for CORS-strict environments
      const coreBase = "https://unpkg.com/@ffmpeg/core@0.12.6/dist/umd";
      let loadOpts = {
        coreURL: `${coreBase}/ffmpeg-core.js`,
        wasmURL: `${coreBase}/ffmpeg-core.wasm`,
      };

      try {
        await ffmpeg.load(loadOpts);
      } catch {
        const { toBlobURL } = window.FFmpegUtil;
        loadOpts = {
          coreURL: await toBlobURL(`${coreBase}/ffmpeg-core.js`,   "text/javascript"),
          wasmURL: await toBlobURL(`${coreBase}/ffmpeg-core.wasm`, "application/wasm"),
        };
        await ffmpeg.load(loadOpts);
      }

      setProgress(10);

      // ── 2. Write each track's audio to virtual FS ───────────────────────
      setStage("processing");
      const total = exportableTracks.length;

      for (let i = 0; i < total; i++) {
        const t = exportableTracks[i];
        setStatus(`READING TRACK ${i + 1}/${total}: ${t.name}…`);
        const blobUrl = BLOB_URLS.get(t.id);
        const data    = await fetchFile(blobUrl);
        await ffmpeg.writeFile(`in_${i}`, data);
        setProgress(10 + Math.round(((i + 1) / total) * 30));
      }

      // ── 3. Build filter_complex ─────────────────────────────────────────
      setStatus("BUILDING FILTER GRAPH…");
      setProgress(42);

      const inputs       = [];
      const filterParts  = [];

      exportableTracks.forEach((t, i) => {
        inputs.push("-i", `in_${i}`);

        const vol         = (t.volume / 100).toFixed(4);
        const activeClips = (t.clips || []).filter((c) => c.active);

        if (activeClips.length > 0) {
          const trimStart  = activeClips[0].audioOffset.toFixed(4);
          const clipDur    = activeClips.reduce((s, c) => s + (c.end - c.start), 0);
          const trimEnd    = (parseFloat(trimStart) + clipDur).toFixed(4);
          filterParts.push(
            `[${i}:a]atrim=start=${trimStart}:end=${trimEnd},asetpts=PTS-STARTPTS,volume=${vol}[a${i}]`
          );
        } else {
          filterParts.push(`[${i}:a]volume=${vol}[a${i}]`);
        }
      });

      let filterComplex;
      if (total === 1) {
        // Single track: rename [a0] → [out] directly
        filterComplex = filterParts[0].replace(/\[a0\]$/, "[out]");
      } else {
        const mixIns  = exportableTracks.map((_, i) => `[a${i}]`).join("");
        filterComplex = [
          ...filterParts,
          `${mixIns}amix=inputs=${total}:normalize=0:dropout_transition=0[out]`,
        ].join(";");
      }

      // ── 4. Run FFmpeg ───────────────────────────────────────────────────
      setStatus("RENDERING MIX…");
      setProgress(50);

      const outFile = `export.${format}`;
      const ffArgs  = [...inputs, "-filter_complex", filterComplex, "-map", "[out]"];

      if (format === "wav") {
        ffArgs.push("-ar", "44100", "-acodec", "pcm_s16le", outFile);
      } else {
        ffArgs.push("-ar", "44100", "-acodec", "libmp3lame", "-q:a", "2", outFile);
      }

      await ffmpeg.exec(ffArgs);

      // ── 5. Read output → trigger download ──────────────────────────────
      setStatus("PREPARING DOWNLOAD…");
      setProgress(95);

      const outData  = await ffmpeg.readFile(outFile);
      const mimeType = format === "wav" ? "audio/wav" : "audio/mpeg";
      const blob     = new Blob([outData.buffer], { type: mimeType });
      const url      = URL.createObjectURL(blob);

      const a = document.createElement("a");
      a.href     = url;
      a.download = `alone_studio_mix.${format}`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      setTimeout(() => URL.revokeObjectURL(url), 8000);

      setProgress(100);
      setStatus("EXPORT COMPLETE ✓");
      setStage("done");

    } catch (err) {
      console.error("[ExportModal] FFmpeg error:", err);
      setError(err?.message || String(err));
      setStage("error");
    }
  };

  const STAGE_LABEL = {
    "idle":         "",
    "loading-core": "LOADING WASM ENGINE",
    "processing":   "PROCESSING AUDIO",
    "done":         "EXPORT COMPLETE",
    "error":        "EXPORT FAILED",
  };

  const accentColor = stage === "error" ? C.pink : C.cyan;

  return (
    <motion.div
      initial={{ opacity:0, scale:0.92, y:24 }}
      animate={{ opacity:1, scale:1,    y:0  }}
      exit={   { opacity:0, scale:0.92, y:24 }}
      style={{ background:C.bgPanel, border:`1px solid ${accentColor}`, padding:26,
        width:420, position:"relative",
        boxShadow:`0 0 70px rgba(${rgb(accentColor)},0.12),0 24px 64px rgba(0,0,0,0.75)` }}>

      <CornerDeco pos="top-left"/>   <CornerDeco pos="top-right"/>
      <CornerDeco pos="bottom-left"/><CornerDeco pos="bottom-right"/>

      {/* ── Header ───────────────────────────────────────────────────────── */}
      <div style={{ display:"flex",alignItems:"center",justifyContent:"space-between",marginBottom:20 }}>
        <div style={{ display:"flex",alignItems:"center",gap:10 }}>
          <motion.div
            animate={{ opacity: isBusy ? [1,0.2,1] : 1 }}
            transition={{ duration:0.55,repeat:Infinity }}
            style={{ width:6,height:6,background:accentColor,boxShadow:`0 0 10px ${accentColor}` }}/>
          <span style={{ fontFamily:"monospace",fontSize:10,color:accentColor,letterSpacing:4 }}>
            EXPORT STUDIO
          </span>
        </div>
        <button onClick={onClose}
          style={{ background:"none",border:"none",color:C.textDim,cursor:"pointer",fontSize:18,lineHeight:1 }}>✕</button>
      </div>

      {/* ── Track list ───────────────────────────────────────────────────── */}
      <div style={{ marginBottom:16,padding:"10px 12px",border:`1px solid ${C.border}`,background:"rgba(0,0,0,0.3)" }}>
        <div style={{ display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:6 }}>
          <span style={{ fontFamily:"monospace",fontSize:9,color:C.textDim,letterSpacing:2 }}>TRACKS QUEUED</span>
          <span style={{ fontFamily:"monospace",fontSize:14,color:C.cyan }}>{exportableTracks.length}</span>
        </div>

        {exportableTracks.length === 0 ? (
          <div style={{ fontFamily:"monospace",fontSize:8,color:C.pink,marginTop:2 }}>
            NO LOADED AUDIO FOUND — DROP FILES ONTO TRACKS FIRST
          </div>
        ) : (
          exportableTracks.map((t) => {
            const activeClips = (t.clips || []).filter((c) => c.active);
            const totalDur    = activeClips.reduce((s, c) => s + (c.end - c.start), 0);
            return (
              <div key={t.id} style={{ display:"flex",alignItems:"center",gap:6,marginTop:5 }}>
                <div style={{ width:5,height:5,borderRadius:1,background:t.color,boxShadow:`0 0 4px ${t.color}`,flexShrink:0 }}/>
                <span style={{ fontFamily:"monospace",fontSize:8,color:C.textBright,flex:1,overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap" }}>
                  {t.name}
                </span>
                <span style={{ fontFamily:"monospace",fontSize:8,color:C.textDim,flexShrink:0 }}>
                  {activeClips.length} CLIP{activeClips.length !== 1 ? "S" : ""} · {totalDur > 0 ? fmtTime(totalDur).split(".")[0] : fmtTime(t.duration).split(".")[0]}
                </span>
                <span style={{ fontFamily:"monospace",fontSize:8,color:t.color,flexShrink:0,minWidth:32,textAlign:"right" }}>
                  {t.volume}%
                </span>
              </div>
            );
          })
        )}
      </div>

      {/* ── Format picker ────────────────────────────────────────────────── */}
      <div style={{ marginBottom:16 }}>
        <div style={{ fontFamily:"monospace",fontSize:8,color:C.textDim,letterSpacing:3,marginBottom:8 }}>OUTPUT FORMAT</div>
        <div style={{ display:"flex",gap:8 }}>
          <CyberButton small active={format==="wav"} color={C.cyan} disabled={isBusy}
            onClick={() => setFormat("wav")}>WAV · LOSSLESS</CyberButton>
          <CyberButton small active={format==="mp3"} color={C.cyan} disabled={isBusy}
            onClick={() => setFormat("mp3")}>MP3 · COMPRESSED</CyberButton>
        </div>
      </div>

      {/* ── FFmpeg status dot ────────────────────────────────────────────── */}
      <div style={{ marginBottom:16,display:"flex",alignItems:"center",gap:6 }}>
        <div style={{ width:5,height:5,borderRadius:"50%",flexShrink:0,
          background: libsReady.ffmpeg ? C.cyan : C.border,
          boxShadow:  libsReady.ffmpeg ? `0 0 6px ${C.cyan}` : "none" }}/>
        <span style={{ fontFamily:"monospace",fontSize:8,color: libsReady.ffmpeg ? C.textBright : C.textDim }}>
          FFMPEG.WASM JS {libsReady.ffmpeg ? "READY" : "LOADING…"}
        </span>
        <span style={{ fontFamily:"monospace",fontSize:7,color:C.textDim,marginLeft:"auto" }}>
          WASM CORE LOADS ON FIRST EXPORT
        </span>
      </div>

      {/* ── Progress bar ─────────────────────────────────────────────────── */}
      {isBusy && (
        <div style={{ marginBottom:16 }}>
          <div style={{ display:"flex",justifyContent:"space-between",marginBottom:5 }}>
            <span style={{ fontFamily:"monospace",fontSize:8,color:C.textDim,flex:1,overflow:"hidden",
              whiteSpace:"nowrap",textOverflow:"ellipsis",maxWidth:300 }}>
              {statusMsg}
            </span>
            <span style={{ fontFamily:"monospace",fontSize:8,color:C.cyan,flexShrink:0 }}>{progress}%</span>
          </div>
          <div style={{ height:3,background:C.border,borderRadius:2,overflow:"hidden" }}>
            <motion.div animate={{ width:`${progress}%` }} transition={{ duration:0.3 }}
              style={{ height:"100%",background:`linear-gradient(90deg,${C.cyan},${C.purple})`,borderRadius:2 }}/>
          </div>
          <div style={{ fontFamily:"monospace",fontSize:7,color:C.textDim,marginTop:4,letterSpacing:2 }}>
            {STAGE_LABEL[stage]}
          </div>
        </div>
      )}

      {/* ── Done state ───────────────────────────────────────────────────── */}
      {stage === "done" && (
        <motion.div initial={{ opacity:0 }} animate={{ opacity:1 }}
          style={{ marginBottom:16,padding:"10px 12px",border:`1px solid ${C.cyan}`,background:`rgba(0,243,255,0.05)` }}>
          <div style={{ display:"flex",alignItems:"center",gap:10 }}>
            <span style={{ color:C.cyan,fontSize:16 }}>✓</span>
            <div>
              <div style={{ fontFamily:"monospace",fontSize:9,color:C.cyan }}>MIX EXPORTED SUCCESSFULLY</div>
              <div style={{ fontFamily:"monospace",fontSize:8,color:C.textDim,marginTop:2 }}>
                alone_studio_mix.{format} · {exportableTracks.length} TRACK{exportableTracks.length !== 1 ? "S" : ""} · CLIENT-SIDE
              </div>
            </div>
          </div>
        </motion.div>
      )}

      {/* ── Error state ──────────────────────────────────────────────────── */}
      {stage === "error" && (
        <div style={{ marginBottom:16,padding:"10px 12px",border:`1px solid ${C.pink}`,background:`rgba(255,0,60,0.05)` }}>
          <div style={{ fontFamily:"monospace",fontSize:9,color:C.pink,marginBottom:4 }}>EXPORT FAILED</div>
          <div style={{ fontFamily:"monospace",fontSize:8,color:C.textDim,wordBreak:"break-word" }}>{errorMsg}</div>
        </div>
      )}

      {/* ── Action button ────────────────────────────────────────────────── */}
      <CyberButton
        active={!isBusy && canExport}
        color={stage === "error" ? C.pink : C.cyan}
        disabled={isBusy || (!canExport && stage === "idle")}
        onClick={isBusy ? undefined : (stage === "done" || stage === "error") ? onClose : runExport}
        style={{ width:"100%",textAlign:"center",justifyContent:"center" }}>
        {stage === "idle"         && `⬇  EXPORT ${format.toUpperCase()} · ${exportableTracks.length} TRACK${exportableTracks.length !== 1 ? "S" : ""}`}
        {stage === "loading-core" && "⟳  LOADING FFMPEG WASM CORE…"}
        {stage === "processing"   && "⟳  PROCESSING…"}
        {stage === "done"         && "✓  DONE — CLOSE"}
        {stage === "error"        && "✕  CLOSE"}
      </CyberButton>

      <div style={{ marginTop:10,fontFamily:"monospace",fontSize:7,color:C.textDim,textAlign:"center",lineHeight:1.7 }}>
        100% CLIENT-SIDE · ZERO UPLOADS · POWERED BY FFMPEG.WASM
      </div>
    </motion.div>
  );
}

// ═══════════════════════════════════════════════════════════════════════════════
//  TRACK ROW  — WaveSurfer waveform + Tone chain + ClipCanvas overlay
// ═══════════════════════════════════════════════════════════════════════════════
function TrackRow({ track, playheadTime, onUpdate, onDelete, selected, onSelect,
                    onOpenEQ, onCut, onSelectClip, selectedClipId, tool, libsReady }) {

  const wsCtnRef  = useRef(null);
  const wsRef     = useRef(null);
  const fileInRef = useRef(null);

  // ── Build Tone chain once libs are ready ────────────────────────────────────
  useEffect(() => {
    if (!libsReady.audio) return;
    buildChain(track.id);
    return () => {
      teardown(track.id);
      const url = BLOB_URLS.get(track.id);
      if (url) { URL.revokeObjectURL(url); BLOB_URLS.delete(track.id); }
    };
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [libsReady.audio]);

  // ── Sync volume + mute ──────────────────────────────────────────────────────
  useEffect(() => {
    const n = getNodes(track.id);
    if (!n) return;
    const db = volToDb(track.muted ? 0 : track.volume);
    n.vol.volume.value = isFinite(db) ? db : -100;
  }, [track.volume, track.muted, track.id]);

  // ── Sync EQ gains ────────────────────────────────────────────────────────────
  useEffect(() => {
    const n = getNodes(track.id);
    if (!n?.eq10) return;
    track.eq.forEach((g, i) => { try { n.eq10[EQ_KEYS[i]].value = g; } catch {} });
  }, [track.eq, track.id]);

  // ── Create WaveSurfer instance ───────────────────────────────────────────────
  useEffect(() => {
    if (!libsReady.audio || !wsCtnRef.current || wsRef.current) return;
    wsRef.current = window.WaveSurfer.create({
      container:     wsCtnRef.current,
      height:        52,
      waveColor:     `rgba(${rgb(track.color)},0.45)`,
      progressColor: track.color,
      cursorColor:   C.cyan,
      cursorWidth:   1.5,
      barWidth:      2,
      barGap:        1,
      barRadius:     2,
      normalize:     true,
      interact:      false,  // We handle clicks via ClipCanvas
      hideScrollbar: true,
    });
    return () => { try { wsRef.current?.destroy(); } catch {} wsRef.current = null; };
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [libsReady.audio]);

  // ── Update WaveSurfer colours when track colour changes ─────────────────────
  useEffect(() => {
    if (!wsRef.current) return;
    try {
      wsRef.current.setOptions({
        waveColor:     `rgba(${rgb(track.color)},0.45)`,
        progressColor: track.color,
      });
    } catch {}
  }, [track.color]);

  // ── Sync playhead → WaveSurfer cursor (visual only) ─────────────────────────
  useEffect(() => {
    if (!wsRef.current || !track.duration) return;
    const p = Math.min(playheadTime / track.duration, 1);
    try { wsRef.current.seekTo(p); } catch {}
  }, [playheadTime, track.duration]);

  // ── File loader (shared by drag-drop + browse) ───────────────────────────────
  const loadFile = useCallback(async (file) => {
    if (!file || !libsReady.audio) return;

    const oldUrl = BLOB_URLS.get(track.id);
    if (oldUrl) URL.revokeObjectURL(oldUrl);

    const url = URL.createObjectURL(file);
    BLOB_URLS.set(track.id, url);

    // 1 — WaveSurfer visual
    if (wsRef.current) {
      try { await wsRef.current.load(url); } catch (e) { console.warn("WS load:", e); }
    }

    // 2 — Tone.Player (actual audio)
    const n = getNodes(track.id);
    if (n?.player) {
      try {
        await n.player.load(url);
        const dur = n.player.buffer?.duration ?? 0;
        onUpdate({
          loaded:   true,
          fileName: file.name,
          duration: dur,
          // ── Initialize single full-length clip ──
          clips: [{ id: newClipId(), start: 0, end: dur, audioOffset: 0, active: true }],
        });
      } catch (e) {
        console.error("Tone load:", e);
        onUpdate({ loaded: true, fileName: file.name, duration: 0, clips: [] });
      }
    }
  }, [libsReady.audio, track.id, onUpdate]);

  // ── Consume PENDING file (from global drop zone) ─────────────────────────────
  useEffect(() => {
    const f = PENDING.get(track.id);
    if (f) { PENDING.delete(track.id); loadFile(f); }
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [track.id]);

  const onFileDrop   = useCallback((e) => {
    e.preventDefault(); e.stopPropagation();
    const f = e.dataTransfer?.files?.[0];
    if (f?.type.startsWith("audio/")) loadFile(f);
  }, [loadFile]);

  const onFileBrowse = useCallback((e) => {
    const f = e.target.files?.[0];
    if (f) loadFile(f);
  }, [loadFile]);

  const fmtDur = (s) => !s ? "--:--" : `${Math.floor(s/60)}:${String(Math.floor(s%60)).padStart(2,"0")}`;

  return (
    <motion.div layout
      initial={{ opacity:0,x:-24 }} animate={{ opacity:1,x:0 }} exit={{ opacity:0,x:-24 }}
      style={{ display:"flex",borderBottom:`1px solid ${C.border}`,position:"relative",
               background: selected ? `rgba(${rgb(track.color)},0.03)` : "transparent" }}
      onDragOver={(e) => e.preventDefault()} onDrop={onFileDrop}>

      {/* ── TRACK HEADER ─────────────────────────────────────────────────── */}
      <div style={{ width:180,minWidth:180,padding:"8px 10px",display:"flex",flexDirection:"column",
                    gap:5,borderRight:`1px solid ${C.border}`,background:C.bgPanel }}>

        {/* Name row */}
        <div style={{ display:"flex",alignItems:"center",justifyContent:"space-between" }}>
          <div style={{ display:"flex",alignItems:"center",gap:6 }}>
            <div style={{ width:8,height:8,borderRadius:1,flexShrink:0,cursor:"pointer",
              background: track.muted ? C.border : track.color,
              boxShadow:  !track.muted ? `0 0 6px ${track.color}` : "none",
              transition:"all 0.2s" }}
              onClick={() => onUpdate({ muted:!track.muted })}/>
            <span style={{ fontFamily:"monospace",fontSize:10,letterSpacing:2,
              color: track.muted ? C.textDim : C.textBright,
              overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap",maxWidth:104 }}>
              {track.name}
            </span>
          </div>
          {track.armed && (
            <motion.div animate={{ opacity:[1,0.15,1] }} transition={{ duration:0.75,repeat:Infinity }}
              style={{ width:6,height:6,borderRadius:"50%",flexShrink:0,
                background:C.pink,boxShadow:`0 0 6px ${C.pink}` }}/>
          )}
        </div>

        {/* Controls row */}
        <div style={{ display:"flex",alignItems:"center",gap:3 }}>
          <CyberButton small active={track.muted}  danger onClick={() => onUpdate({ muted:!track.muted })}>M</CyberButton>
          <CyberButton small active={track.solo}   color={C.yellow} onClick={() => onUpdate({ solo:!track.solo })}>S</CyberButton>
          <CyberButton small active={track.armed}  danger onClick={() => onUpdate({ armed:!track.armed })}>R</CyberButton>
          <CyberButton small color={C.cyan}        onClick={onOpenEQ}>EQ</CyberButton>
          <motion.button whileHover={{ color:C.pink }} onClick={onDelete}
            style={{ background:"none",border:"none",color:C.textDim,cursor:"pointer",
                     fontSize:10,fontFamily:"monospace",marginLeft:"auto",padding:"0 2px" }}>✕</motion.button>
        </div>

        {/* Volume */}
        <div style={{ display:"flex",alignItems:"center",gap:6 }}>
          <span style={{ fontSize:8,fontFamily:"monospace",color:C.textDim }}>VOL</span>
          <div style={{ flex:1 }}>
            <NeonSlider value={track.volume} color={track.color} onChange={(v) => onUpdate({ volume:v })}/>
          </div>
          <span style={{ fontSize:8,fontFamily:"monospace",color:track.color,width:22,textAlign:"right" }}>
            {track.volume}
          </span>
        </div>

        {/* VU meter */}
        <VUMeter trackId={track.id} active={track.loaded && !track.muted} color={track.color}/>

        {/* File info */}
        <div style={{ fontSize:7,fontFamily:"monospace",color:C.textDim,
                      overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap" }}>
          {track.loaded
            ? `${track.fileName.slice(0,18)} · ${fmtDur(track.duration)}`
            : "NO FILE LOADED"}
        </div>

        {/* Clip count badge */}
        {track.loaded && track.clips.length > 1 && (
          <div style={{ fontSize:7,fontFamily:"monospace",color:C.cyan,letterSpacing:1 }}>
            ✂ {track.clips.filter((c) => c.active).length}/{track.clips.length} CLIPS ACTIVE
          </div>
        )}
      </div>

      {/* ── WAVEFORM + CLIP CANVAS ────────────────────────────────────────── */}
      <div style={{ flex:1,position:"relative",overflow:"hidden",height:52 }} onClick={onSelect}>

        {/* WaveSurfer waveform */}
        <div ref={wsCtnRef} style={{ width:"100%",height:52 }}/>

        {/* Subtle vertical grid */}
        <div style={{ position:"absolute",inset:0,pointerEvents:"none",
          backgroundImage:`repeating-linear-gradient(90deg,transparent,transparent 79px,rgba(255,255,255,0.02) 80px)` }}/>

        {/* Selection ring */}
        {selected && (
          <div style={{ position:"absolute",inset:0,pointerEvents:"none",
            background:`rgba(${rgb(track.color)},0.04)`,
            border:`1px solid rgba(${rgb(track.color)},0.25)` }}/>
        )}

        {/* ─── [Step 4] CLIP CANVAS OVERLAY ──────────────────────────────── */}
        <ClipCanvas
          clips={track.clips}
          duration={track.duration}
          color={track.color}
          tool={tool}
          onCut={(time) => onCut(track.id, time)}
          onSelectClip={(clipId) => onSelectClip(track.id, clipId)}
          selectedClipId={selectedClipId?.trackId === track.id ? selectedClipId.clipId : null}
        />

        {/* Empty-state prompt */}
        {!track.loaded && (
          <div style={{ position:"absolute",inset:0,display:"flex",alignItems:"center",
                        justifyContent:"center",gap:10,pointerEvents:"none" }}>
            <motion.span animate={{ opacity:[0.3,0.7,0.3] }} transition={{ duration:2.5,repeat:Infinity }}
              style={{ fontFamily:"monospace",fontSize:9,color:C.textDim,letterSpacing:3 }}>
              DROP AUDIO  ·  OR
            </motion.span>
            <motion.button
              whileHover={{ color:C.cyan,borderColor:C.cyan }}
              onClick={(e) => { e.stopPropagation(); fileInRef.current?.click(); }}
              style={{ background:"none",border:`1px solid ${C.border}`,color:C.textDim,
                       fontFamily:"monospace",fontSize:9,letterSpacing:3,padding:"2px 10px",
                       cursor:"pointer",pointerEvents:"auto" }}>
              BROWSE
            </motion.button>
          </div>
        )}

        <input ref={fileInRef} type="file" accept="audio/*" style={{ display:"none" }} onChange={onFileBrowse}/>
      </div>
    </motion.div>
  );
}

//  ROOT APP
// ═══════════════════════════════════════════════════════════════════════════════
export default function AloneAudioStudio() {
  const libsReady = useCDNLibs();

  const [tracks,        setTracks]        = useState([]);
  const [isPlaying,     setIsPlaying]     = useState(false);
  const [playheadTime,  setPlayheadTime]  = useState(0);
  const [tool,          setTool]          = useState("select");
  const [selectedTrack, setSelectedTrack] = useState(null);
  const [selectedClip,  setSelectedClip]  = useState(null); // { trackId, clipId }
  const [showStem,      setShowStem]      = useState(false);
  const [showEQ,        setShowEQ]        = useState(false);
  const [showExport,    setShowExport]    = useState(false);
  const [eqTargetId,    setEqTargetId]    = useState(null);
  const [bpm,           setBpm]           = useState(128);
  const [masterVol,     setMasterVol]     = useState(85);
  const [isDragging,    setIsDragging]    = useState(false);

  const phtRef    = useRef(0);
  const tracksRef = useRef(tracks); // always-fresh reference, avoids stale closures
  useEffect(() => { tracksRef.current = tracks; }, [tracks]);

  // ── Master volume → Tone.Destination ────────────────────────────────────────
  useEffect(() => {
    if (!libsReady.audio || !window.Tone) return;
    try { window.Tone.getDestination().volume.value = volToDb(masterVol); } catch {}
  }, [masterVol, libsReady.audio]);

  // ── [Step 4] TRANSPORT-BASED PLAYBACK ───────────────────────────────────────
  // Schedules every active clip from every non-muted loaded track on Tone.Transport.
  // Only triggered when isPlaying toggles — edits take effect on next press of play.
  useEffect(() => {
    if (!libsReady.audio || !window.Tone) return;
    const T = window.Tone;

    if (isPlaying) {
      T.Transport.stop();
      T.Transport.cancel();
      T.Transport.seconds = phtRef.current; // resume from current position

      const currentTracks = tracksRef.current;

      currentTracks.forEach((track) => {
        if (track.muted || !track.loaded) return;
        const n = getNodes(track.id);
        if (!n?.player?.loaded) return;

        const activeClips = (track.clips || []).filter((c) => c.active && c.end > c.start);

        if (activeClips.length === 0) {
          // No clips defined (shouldn't happen after load, but be safe)
          T.Transport.schedule((t) => {
            try { T.start(); n.player.start(t); } catch {}
          }, 0);
        } else {
          activeClips.forEach((clip) => {
            const bufDur = n.player.buffer?.duration ?? 0;
            if (clip.audioOffset >= bufDur) return;
            const playDur = Math.min(clip.end - clip.start, bufDur - clip.audioOffset);
            if (playDur <= 0.01) return;

            T.Transport.schedule((t) => {
              try { T.start(); n.player.start(t, clip.audioOffset, playDur); } catch {}
            }, Math.max(0, clip.start));
          });
        }
      });

      T.Transport.start();

    } else {
      // Stop transport + all players
      if (window.Tone) {
        try { window.Tone.Transport.pause(); } catch {}
      }
      NODES.forEach((n) => { try { n.player.stop(); } catch {} });
    }
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [isPlaying, libsReady.audio]);

  // ── Playhead: read from Tone.Transport while playing ────────────────────────
  useEffect(() => {
    if (!isPlaying || !libsReady.audio) return;
    const id = setInterval(() => {
      if (!window.Tone) return;
      const t = window.Tone.Transport.seconds;
      phtRef.current = t;
      setPlayheadTime(t);
    }, 33);
    return () => clearInterval(id);
  }, [isPlaying, libsReady.audio]);

  const handleStop = useCallback(() => {
    setIsPlaying(false);
    if (window.Tone) {
      try { window.Tone.Transport.stop(); window.Tone.Transport.seconds = 0; } catch {}
    }
    phtRef.current = 0;
    setPlayheadTime(0);
  }, []);

  // ── Track CRUD ───────────────────────────────────────────────────────────────
  const addTrack    = useCallback((o = {}) => setTracks((ts) => [...ts, makeTrack(o)]), []);
  const updateTrack = useCallback((id, p) => setTracks((ts) => ts.map((t) => t.id===id ? {...t,...p} : t)), []);
  const deleteTrack = useCallback((id) => {
    teardown(id);
    const url = BLOB_URLS.get(id);
    if (url) { URL.revokeObjectURL(url); BLOB_URLS.delete(id); }
    setTracks((ts) => ts.filter((t) => t.id !== id));
  }, []);

  const updateEQ = useCallback((trackId, band, gain) => {
    setTracks((ts) => ts.map((t) => {
      if (t.id !== trackId) return t;
      const eq = [...t.eq]; eq[band] = gain; return { ...t, eq };
    }));
  }, []);

  // ── [Step 4] Cut & clip selection ───────────────────────────────────────────
  const handleCut = useCallback((trackId, cutTime) => {
    setTracks((ts) => ts.map((t) => {
      if (t.id !== trackId || !t.duration) return t;
      return { ...t, clips: cutClipAt(t.clips, cutTime) };
    }));
  }, []);

  const handleSelectClip = useCallback((trackId, clipId) => {
    setSelectedClip(clipId ? { trackId, clipId } : null);
    // Also select the track
    setSelectedTrack(trackId);
  }, []);

  // Delete key → remove selected clip
  useEffect(() => {
    const h = (e) => {
      if (e.target.tagName === "INPUT") return;
      if ((e.key === "Delete" || e.key === "Backspace") && selectedClip) {
        setTracks((ts) => ts.map((t) => {
          if (t.id !== selectedClip.trackId) return t;
          return { ...t, clips: t.clips.map((c) =>
            c.id === selectedClip.clipId ? { ...c, active: false } : c
          )};
        }));
        setSelectedClip(null);
      }
    };
    window.addEventListener("keydown", h);
    return () => window.removeEventListener("keydown", h);
  }, [selectedClip]);

  // ── Stem AI callback ─────────────────────────────────────────────────────────
  const handleStemDone = useCallback(() => {
    [
      { name:"VOCALS•AI", color:C.cyan   },
      { name:"BASS•AI",   color:C.purple },
      { name:"DRUMS•AI",  color:C.pink   },
      { name:"MELODY•AI", color:C.yellow },
    ].forEach((o) => addTrack(o));
  }, [addTrack]);

  // ── Global drop zone ─────────────────────────────────────────────────────────
  const handleGlobalDrop = useCallback((e) => {
    e.preventDefault();
    setIsDragging(false);
    const files = [...(e.dataTransfer?.files || [])].filter((f) => f.type.startsWith("audio/"));
    files.forEach((f) => {
      const t = makeTrack({ name: f.name.replace(/\.[^.]+$/, "").toUpperCase().slice(0, 16) });
      PENDING.set(t.id, f);
      setTracks((ts) => [...ts, t]);
    });
  }, []);

  // ── EQ helpers ───────────────────────────────────────────────────────────────
  const eqTrack = useMemo(() => tracks.find((t) => t.id === eqTargetId), [tracks, eqTargetId]);
  const openEQ  = useCallback((id) => { setEqTargetId(id); setShowEQ(true); }, []);
  const openEQDefault = useCallback(() => {
    if (!tracks.length) return;
    openEQ(selectedTrack ?? tracks[0].id);
  }, [tracks, selectedTrack, openEQ]);

  // ── Keyboard shortcuts ───────────────────────────────────────────────────────
  useEffect(() => {
    const h = (e) => {
      if (e.target.tagName === "INPUT") return;
      if (e.code === "Space")  { e.preventDefault(); setIsPlaying((p) => !p); }
      if (e.key  === "Escape") handleStop();
      if (e.key  === "c" || e.key === "C") setTool("cut");
      if (e.key  === "v" || e.key === "V") setTool("select");
      if (e.key  === "d" || e.key === "D") setTool("draw");
    };
    window.addEventListener("keydown", h);
    return () => window.removeEventListener("keydown", h);
  }, [handleStop]);

  const maxDuration = useMemo(() => Math.max(60, ...tracks.map((t) => t.duration || 0)), [tracks]);
  const hasAny      = tracks.length > 0;

  // ── Render ───────────────────────────────────────────────────────────────────
  return (
    <div style={{ background:C.bg, height:"100vh", display:"flex", flexDirection:"column",
                  fontFamily:"monospace", color:C.text, overflow:"hidden", position:"relative" }}>
      <Scanlines/>

      {/* Ambient glows */}
      <div style={{ position:"fixed",inset:0,zIndex:0,overflow:"hidden",pointerEvents:"none" }}>
        <div style={{ position:"absolute",top:-200,left:-200,width:600,height:600,borderRadius:"50%",
          background:`radial-gradient(circle,rgba(0,243,255,0.04) 0%,transparent 70%)` }}/>
        <div style={{ position:"absolute",bottom:-200,right:-200,width:600,height:600,borderRadius:"50%",
          background:`radial-gradient(circle,rgba(255,0,60,0.04) 0%,transparent 70%)` }}/>
      </div>

      {/* ── Lib loading splash ─────────────────────────────────────────────── */}
      <AnimatePresence>
        {!libsReady.audio && (
          <motion.div initial={{ opacity:0 }} animate={{ opacity:1 }} exit={{ opacity:0 }}
            style={{ position:"absolute",inset:0,zIndex:200,background:"rgba(4,4,10,0.97)",
                     display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",gap:20 }}>
            <motion.div animate={{ opacity:[1,0.25,1] }} transition={{ duration:1.1,repeat:Infinity }}
              style={{ fontFamily:"monospace",fontSize:11,color:C.cyan,letterSpacing:6 }}>
              LOADING AUDIO ENGINE
            </motion.div>
            <div style={{ display:"flex",gap:8 }}>
              {["WAVESURFER 7","TONE.JS 14","WEB AUDIO API"].map((n) => (
                <span key={n} style={{ fontFamily:"monospace",fontSize:8,color:C.textDim,
                  border:`1px solid ${C.border}`,padding:"2px 10px" }}>{n}</span>
              ))}
            </div>
            <motion.div animate={{ width:["0%","70%","0%"] }} transition={{ duration:2,repeat:Infinity }}
              style={{ height:1,background:`linear-gradient(90deg,transparent,${C.cyan},transparent)` }}/>
          </motion.div>
        )}
      </AnimatePresence>

      {/* ── Toolbar ─────────────────────────────────────────────────────────── */}
      <Toolbar tool={tool} setTool={setTool} isPlaying={isPlaying}
        onPlayPause={() => setIsPlaying((p) => !p)} onStop={handleStop}
        onAddTrack={() => addTrack()} onShowStem={() => setShowStem(true)}
        onShowEQ={openEQDefault} onExport={() => setShowExport(true)}
        bpm={bpm} setBpm={setBpm}
        playheadTime={playheadTime} hasAny={hasAny} libsReady={libsReady}/>

      {/* ── Main workspace ───────────────────────────────────────────────────── */}
      <div style={{ display:"flex",flex:1,overflow:"hidden",position:"relative",zIndex:1 }}>

        {/* Timeline column */}
        <div style={{ flex:1,display:"flex",flexDirection:"column",overflow:"hidden" }}>
          <TimelineRuler maxDuration={maxDuration}/>

          {/* Drop + scroll zone */}
          <div style={{ flex:1,overflowY:"auto",overflowX:"hidden",position:"relative" }}
            onDragOver={(e) => { e.preventDefault(); setIsDragging(true); }}
            onDragLeave={() => setIsDragging(false)}
            onDrop={handleGlobalDrop}>

            {/* Drop overlay */}
            <AnimatePresence>
              {isDragging && (
                <motion.div initial={{ opacity:0 }} animate={{ opacity:1 }} exit={{ opacity:0 }}
                  style={{ position:"absolute",inset:0,zIndex:40,background:"rgba(0,4,10,0.88)",
                    border:`2px dashed ${C.cyan}`,display:"flex",alignItems:"center",justifyContent:"center",
                    boxShadow:`inset 0 0 80px rgba(0,243,255,0.06)`,pointerEvents:"none" }}>
                  <div style={{ textAlign:"center" }}>
                    <motion.div animate={{ scale:[1,1.1,1] }} transition={{ duration:1.1,repeat:Infinity }}
                      style={{ fontSize:44,color:C.cyan,marginBottom:10 }}>⊕</motion.div>
                    <div style={{ fontFamily:"monospace",fontSize:13,color:C.cyan,letterSpacing:5 }}>DROP AUDIO FILES</div>
                    <div style={{ fontFamily:"monospace",fontSize:9,color:C.textDim,letterSpacing:3,marginTop:6 }}>
                      MP3  ·  WAV  ·  FLAC  ·  OGG  ·  M4A
                    </div>
                  </div>
                </motion.div>
              )}
            </AnimatePresence>

            {/* Empty state */}
            {tracks.length === 0 && (
              <motion.div initial={{ opacity:0 }} animate={{ opacity:1 }}
                style={{ padding:56,display:"flex",flexDirection:"column",alignItems:"center",gap:18 }}>
                <motion.div animate={{ opacity:[0.3,0.7,0.3] }} transition={{ duration:3.5,repeat:Infinity }}
                  style={{ fontSize:52,color:C.border }}>⟨⟩</motion.div>
                <div style={{ fontFamily:"monospace",fontSize:11,color:C.textDim,letterSpacing:6 }}>NO TRACKS</div>
                <div style={{ fontFamily:"monospace",fontSize:9,color:C.textDim,letterSpacing:3 }}>
                  DRAG & DROP AUDIO FILES  ·  OR  ·  ADD A TRACK BELOW
                </div>
                <CyberButton active color={C.cyan} onClick={() => addTrack()}>
                  ⊕  ADD YOUR FIRST TRACK
                </CyberButton>
              </motion.div>
            )}

            {/* Track rows */}
            <AnimatePresence>
              {tracks.map((track) => (
                <TrackRow key={track.id} track={track}
                  playheadTime={playheadTime}
                  onUpdate={(p)  => updateTrack(track.id, p)}
                  onDelete={() => deleteTrack(track.id)}
                  selected={selectedTrack === track.id}
                  onSelect={() => setSelectedTrack((id) => id===track.id ? null : track.id)}
                  onOpenEQ={() => openEQ(track.id)}
                  onCut={handleCut}
                  onSelectClip={handleSelectClip}
                  selectedClipId={selectedClip}
                  tool={tool}
                  libsReady={libsReady}/>
              ))}
            </AnimatePresence>

            {/* Add track row */}
            {tracks.length > 0 && (
              <motion.button whileHover={{ background:C.bgHover }} onClick={() => addTrack()}
                style={{ width:"100%",padding:"10px 14px",background:"none",border:"none",
                  borderBottom:`1px solid ${C.border}`,cursor:"pointer",display:"flex",alignItems:"center",gap:8 }}>
                <span style={{ fontSize:16,color:C.textDim }}>+</span>
                <span style={{ fontFamily:"monospace",fontSize:9,color:C.textDim,letterSpacing:3 }}>ADD TRACK</span>
              </motion.button>
            )}
          </div>

          <EffectsRack masterVol={masterVol} onMasterVol={setMasterVol}/>
        </div>

        {/* ── Right panel ──────────────────────────────────────────────────── */}
        <motion.div initial={{ x:220 }} animate={{ x:0 }} transition={{ type:"spring",stiffness:150,damping:26 }}
          style={{ width:220,borderLeft:`1px solid ${C.border}`,background:C.bgPanel,
                   display:"flex",flexDirection:"column",flexShrink:0 }}>

          {/* Master */}
          <div style={{ padding:"12px 14px",borderBottom:`1px solid ${C.border}` }}>
            <div style={{ fontFamily:"monospace",fontSize:9,color:C.cyan,letterSpacing:3,marginBottom:10 }}>MASTER</div>
            <div style={{ display:"flex",alignItems:"center",gap:10 }}>
              <div style={{ flex:1 }}><NeonSlider value={masterVol} color={C.cyan} onChange={setMasterVol}/></div>
              <VUMeter trackId="__master__" active={isPlaying} color={C.cyan}/>
            </div>
            <div style={{ marginTop:8,display:"flex",gap:4 }}>
              <CyberButton small active>MONO</CyberButton>
              <CyberButton small>STEREO</CyberButton>
            </div>
          </div>

          {/* Engine status */}
          <div style={{ padding:"12px 14px",borderBottom:`1px solid ${C.border}` }}>
            <div style={{ fontFamily:"monospace",fontSize:9,color:C.textDim,letterSpacing:3,marginBottom:8 }}>ENGINE STATUS</div>
            {[
              ["WAVESURFER 7",  libsReady.audio,  "WAVEFORM"],
              ["TONE.JS 14",    libsReady.audio,  "PLAYBACK"],
              ["TRANSPORT",     libsReady.audio,  "CLIPS"],
              ["FFMPEG.WASM",   libsReady.ffmpeg, "EXPORT"],
              ["ONNX RT",       false,            "STEM AI"],
            ].map(([name, ok, desc]) => (
              <div key={name} style={{ display:"flex",alignItems:"center",gap:6,marginBottom:5 }}>
                <div style={{ width:5,height:5,borderRadius:"50%",flexShrink:0,
                  background: ok ? C.cyan : C.border,
                  boxShadow:  ok ? `0 0 6px ${C.cyan}` : "none" }}/>
                <span style={{ fontSize:8,fontFamily:"monospace",color: ok ? C.cyan : C.textDim }}>{name}</span>
                <span style={{ fontSize:7,fontFamily:"monospace",color:C.textDim,marginLeft:"auto" }}>{desc}</span>
              </div>
            ))}
          </div>

          {/* Shortcuts */}
          <div style={{ padding:"12px 14px",borderBottom:`1px solid ${C.border}` }}>
            <div style={{ fontFamily:"monospace",fontSize:9,color:C.textDim,letterSpacing:3,marginBottom:8 }}>SHORTCUTS</div>
            {[
              ["SPACE",   "PLAY / PAUSE"],
              ["ESC",     "STOP + REWIND"],
              ["C",       "CUT TOOL ✂"],
              ["V",       "SELECT TOOL"],
              ["D",       "DRAW TOOL"],
              ["DELETE",  "REMOVE CLIP"],
            ].map(([k, v]) => (
              <div key={k} style={{ display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:5 }}>
                <span style={{ fontSize:8,color:C.cyan,background:`rgba(0,243,255,0.07)`,
                  padding:"1px 6px",letterSpacing:1 }}>{k}</span>
                <span style={{ fontSize:8,color:C.textDim }}>{v}</span>
              </div>
            ))}
          </div>

          {/* Cut tips */}
          <div style={{ padding:"12px 14px",borderBottom:`1px solid ${C.border}` }}>
            <div style={{ fontFamily:"monospace",fontSize:9,color:C.textDim,letterSpacing:3,marginBottom:8 }}>CUT WORKFLOW</div>
            {[
              ["1","SELECT CUT [C] TOOL"],
              ["2","CLICK WAVEFORM TO SPLIT"],
              ["3","SELECT [V] + CLICK CLIP"],
              ["4","PRESS DELETE TO MUTE"],
              ["5","EXPORT → CLEAN MIX"],
            ].map(([n, t]) => (
              <div key={n} style={{ display:"flex",gap:6,alignItems:"flex-start",marginBottom:5 }}>
                <span style={{ fontFamily:"monospace",fontSize:7,color:C.cyan,
                  background:`rgba(0,243,255,0.1)`,padding:"1px 5px",flexShrink:0 }}>{n}</span>
                <span style={{ fontFamily:"monospace",fontSize:7,color:C.textDim,lineHeight:1.4 }}>{t}</span>
              </div>
            ))}
          </div>

          {/* Footer */}
          <div style={{ marginTop:"auto",padding:"10px 14px",borderTop:`1px solid ${C.border}` }}>
            <div style={{ fontFamily:"monospace",fontSize:7,color:C.textDim,letterSpacing:2 }}>ALONE AUDIO STUDIO</div>
            <div style={{ fontFamily:"monospace",fontSize:7,color:C.textDim,letterSpacing:2 }}>v3.0.0 · CLIENT-SIDE</div>
            <div style={{ fontFamily:"monospace",fontSize:7,color:C.cyanDim,letterSpacing:2,marginTop:2 }}>
              CUT · TRANSPORT · FFMPEG.WASM
            </div>
          </div>
        </motion.div>
      </div>

      {/* ── Modals ───────────────────────────────────────────────────────────── */}
      <AnimatePresence>
        {showEQ && eqTrack && (
          <motion.div initial={{ opacity:0 }} animate={{ opacity:1 }} exit={{ opacity:0 }}
            style={{ position:"fixed",inset:0,zIndex:60,display:"flex",alignItems:"flex-end",
                     justifyContent:"center",paddingBottom:56,background:"rgba(0,0,0,0.7)" }}
            onClick={(e) => { if (e.target===e.currentTarget) setShowEQ(false); }}>
            <EQPanel trackId={eqTrack.id} trackName={eqTrack.name}
              gains={eqTrack.eq}
              onGainChange={(i, v) => updateEQ(eqTrack.id, i, v)}
              onClose={() => setShowEQ(false)}/>
          </motion.div>
        )}

        {showStem && (
          <motion.div initial={{ opacity:0 }} animate={{ opacity:1 }} exit={{ opacity:0 }}
            style={{ position:"fixed",inset:0,zIndex:60,display:"flex",alignItems:"center",
                     justifyContent:"center",background:"rgba(0,0,0,0.75)" }}
            onClick={(e) => { if (e.target===e.currentTarget) setShowStem(false); }}>
            <StemPanel onClose={() => setShowStem(false)} onProcess={handleStemDone}/>
          </motion.div>
        )}

        {/* ── [Step 5] EXPORT MODAL ──────────────────────────────────────────── */}
        {showExport && (
          <motion.div initial={{ opacity:0 }} animate={{ opacity:1 }} exit={{ opacity:0 }}
            style={{ position:"fixed",inset:0,zIndex:70,display:"flex",alignItems:"center",
                     justifyContent:"center",background:"rgba(0,0,0,0.8)" }}
            onClick={(e) => { if (e.target===e.currentTarget) setShowExport(false); }}>
            <ExportModal
              tracks={tracks}
              libsReady={libsReady}
              onClose={() => setShowExport(false)}/>
          </motion.div>
        )}
      </AnimatePresence>

      {/* ── Global styles ─────────────────────────────────────────────────────── */}
      <style>{`
        * { box-sizing: border-box; }
        @keyframes pulse { 0%,100%{opacity:1} 50%{opacity:0.15} }
        ::-webkit-scrollbar { width:5px; height:5px; }
        ::-webkit-scrollbar-track { background:${C.bg}; }
        ::-webkit-scrollbar-thumb { background:${C.border}; border-radius:2px; }
        ::-webkit-scrollbar-thumb:hover { background:${C.cyanDim}; }
        input[type=range]::-webkit-slider-thumb {
          -webkit-appearance:none; width:12px; height:12px;
          border-radius:2px; background:${C.cyan};
          cursor:pointer; box-shadow:0 0 6px ${C.cyan};
        }
        input[type=number]::-webkit-inner-spin-button,
        input[type=number]::-webkit-outer-spin-button { opacity:0.25; }
        wave { background: transparent !important; }
      `}</style>
    </div>
  );
}
