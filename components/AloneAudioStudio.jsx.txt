  Array.from({ length: len }, (_, i) => {
    const x = i / len;

      Math.sin(x * 47 * seed) * 0.3 +

      Math.sin(x * 89 * seed) * 0.2 +
      (Math.random() - 0.5) * 0.1
    ) * amp;
  });

function VUMeter({ active, color = C.cyan }) {
  const [bars, setBars] = useState(Array(12).fill(0));
  useEffect(() => {
    if (!active) { setBars(Array(12).fill(0)); return; }
    const id = setInterval(() => {
      setBars(prev => prev.map((_, i) => {
        const target = Math.max(0, (Math.random() * 12) - i * 0.5);
        return Math.min(1, target);
      }));
    }, 80);
    return () => clearInterval(id);
  }, [active]);
  return (
    <div className="flex gap-px items-end" style={{ height: 24 }}>
      {bars.map((b, i) => (
        <motion.div key={i} animate={{ height: `${b * 100}%` }} transition={{ duration: 0.08 }}
          style={{ width: 3, minHeight: 2, background: i > 9 ? C.pink : i > 7 ? C.yellow : color, opacity: 0.9 }} />
      ))}
    </div>
  );
}

function WaveformCanvas({ wave, color, progress = 0.3, selected = false, height = 52 }) {
  const canvasRef = useRef(null);
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas || !wave) return;
    const ctx = canvas.getContext("2d");
    const W = canvas.width, H = canvas.height, mid = H / 2;
    ctx.clearRect(0, 0, W, H);
    const playedX = W * progress;

    // Background grid
    ctx.strokeStyle = "rgba(255,255,255,0.03)";
    ctx.lineWidth = 1;
    for (let x = 0; x < W; x += 40) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke(); }

    // Waveform
    const step = W / wave.length;
    for (let i = 0; i < wave.length; i++) {
      const x = i * step;
      const h = Math.abs(wave[i]) * (H * 0.45);
      const played = x < playedX;
      const alpha = played ? 1 : 0.45;
      ctx.fillStyle = played ? color : `rgba(${hexToRgb(color)},${alpha})`;
      ctx.fillRect(x, mid - h, Math.max(1.5, step - 0.5), h * 2);
    }

    // Playhead
    ctx.strokeStyle = C.cyan;
    ctx.lineWidth = 1.5;
    ctx.shadowColor = C.cyan;
    ctx.shadowBlur = 8;
    ctx.beginPath();
    ctx.moveTo(playedX, 0); ctx.lineTo(playedX, H);
    ctx.stroke();
    ctx.shadowBlur = 0;

    // Selection overlay
    if (selected) {
      ctx.fillStyle = "rgba(0,243,255,0.06)";
      ctx.fillRect(W * 0.2, 0, W * 0.3, H);
      ctx.strokeStyle = `rgba(0,243,255,0.4)`;
      ctx.lineWidth = 1;
      ctx.strokeRect(W * 0.2, 0, W * 0.3, H);
    }
  }, [wave, color, progress, selected]);

  return (
    <canvas ref={canvasRef} width={600} height={height}
      style={{ width: "100%", height, display: "block", cursor: "crosshair" }} />
  );
}

        <input type="number" value={bpm} onChange={e => setBpm(Number(e.target.value))}
          style={{ width: 40, background: "none", border: "none", outline: "none",
            fontFamily: "monospace", fontSize: 16, color: C.cyan, textAlign: "center" }} />
      </div>

      {/* Time Display */}
      <div style={{ padding: "4px 10px", border: `1px solid ${C.border}`, background: "rgba(0,0,0,0.3)" }}>
        <motion.span animate={{ opacity: isPlaying ? [1, 0.7, 1] : 1 }} transition={{ duration: 0.5, repeat: Infinity }}
          style={{ fontFamily: "monospace", fontSize: 18, color: C.cyan, letterSpacing: 2, display: "block" }}>
          0:00.000
        </motion.span>
      </div>

      <div style={{ width: 1, height: 32, background: C.border }} />

      {/* Edit Tools */}
      <div className="flex gap-1">
        {tools.map(t => (
          <CyberButton key={t.id} small active={tool === t.id} onClick={() => setTool(t.id)}>
            {t.label}
            <span style={{ color: C.textDim, marginLeft: 4 }}>[{t.key}]</span>
          </CyberButton>
        ))}
      </div>

      <div style={{ width: 1, height: 32, background: C.border }} />

      {/* Actions */}
      <div className="flex gap-2">
        <CyberButton small onClick={onImport} color={C.cyan} active>⊕ IMPORT</CyberButton>
        <CyberButton small onClick={onShowStem} color={C.pink} active>⊛ STEM AI</CyberButton>
        <CyberButton small onClick={onShowEQ} color={C.cyan}>EQ</CyberButton>
      </div>

      <div style={{ marginLeft: "auto", display: "flex", gap: 10, alignItems: "center" }}>
        <div style={{ display: "flex", gap: 2, alignItems: "center" }}>
          {Array(8).fill(0).map((_, i) => (
            <motion.div key={i} animate={isPlaying ? { height: [4, 8 + Math.random() * 10, 4] } : { height: 4 }}
              transition={{ duration: 0.3 + i * 0.05, repeat: Infinity }}
              style={{ width: 3, background: i > 6 ? C.pink : C.cyan, opacity: isPlaying ? 0.9 : 0.3 }} />
          ))}
        </div>
        <CyberButton small color={C.pink} active>⬇ EXPORT</CyberButton>
      </div>
    </motion.div>
  );
}

// ─── MAIN APP ─────────────────────────────────────────────────────────────────
export default function AloneAudioStudio() {
  const [tracks, setTracks] = useState(INITIAL_TRACKS);
  const [isPlaying, setIsPlaying] = useState(false);
  const [tool, setTool] = useState("select");
  const [selectedTrack, setSelectedTrack] = useState(null);
  const [showStem, setShowStem] = useState(false);
  const [showEQ, setShowEQ] = useState(false);
  const [bpm, setBpm] = useState(128);
  const [zoom, setZoom] = useState(1);
  const [isDragging, setIsDragging] = useState(false);

  const updateTrack = useCallback((id, patch) =>
    setTracks(ts => ts.map(t => t.id === id ? { ...t, ...patch } : t)), []);
  const deleteTrack = useCallback((id) =>
    setTracks(ts => ts.filter(t => t.id !== id)), []);

  const handleStemDone = () => {
    const stemDefs = [
      { id: Date.now() + 1, name: "VOCALS•AI", color: C.cyan,   wave: genWave(300, 0.7, 10), muted: false, solo: false, volume: 80, armed: false, loaded: true },
      { id: Date.now() + 2, name: "BASS•AI",   color: C.purple, wave: genWave(300, 0.9, 11), muted: false, solo: false, volume: 75, armed: false, loaded: true },
      { id: Date.now() + 3, name: "DRUMS•AI",  color: C.pink,   wave: genWave(300, 1.0, 12), muted: false, solo: false, volume: 70, armed: false, loaded: true },
      { id: Date.now() + 4, name: "MELODY•AI", color: C.yellow, wave: genWave(300, 0.6, 13), muted: false, solo: false, volume: 65, armed: false, loaded: true },
    ];
    setTracks(ts => [...ts, ...stemDefs]);
  };

  const addTrack = () => {
    const colors = [C.cyan, C.pink, C.purple, C.yellow];
    setTracks(ts => [...ts, {
      id: Date.now(), name: `TRACK ${ts.length + 1}`, color: colors[ts.length % 4],
      wave: genWave(300, 0.7, ts.length + 1), muted: false, solo: false, volume: 70, armed: false, loaded: false,
    }]);
  };

  // Keyboard shortcuts
  useEffect(() => {
    const onKey = (e) => {
      if (e.target.tagName === "INPUT") return;
      if (e.code === "Space") { e.preventDefault(); setIsPlaying(p => !p); }
      if (e.key === "c" || e.key === "C") setTool("cut");
      if (e.key === "v" || e.key === "V") setTool("select");
      if (e.key === "d" || e.key === "D") setTool("draw");
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, []);

  return (
    <div style={{ background: C.bg, minHeight: "100vh", display: "flex", flexDirection: "column",
      fontFamily: "monospace", color: C.text, position: "relative", overflow: "hidden" }}>
      <Scanlines />

      {/* Ambient glow orbs */}
      <div className="pointer-events-none" style={{ position: "fixed", inset: 0, overflow: "hidden", zIndex: 0 }}>
        <div style={{ position: "absolute", top: -200, left: -200, width: 600, height: 600, borderRadius: "50%",
          background: `radial-gradient(circle, rgba(0,243,255,0.04) 0%, transparent 70%)` }} />
        <div style={{ position: "absolute", bottom: -200, right: -200, width: 600, height: 600, borderRadius: "50%",
          background: `radial-gradient(circle, rgba(255,0,60,0.04) 0%, transparent 70%)` }} />
      </div>

      {/* TOOLBAR */}
      <Toolbar tool={tool} setTool={setTool} isPlaying={isPlaying}
        onPlayPause={() => setIsPlaying(p => !p)} onStop={() => setIsPlaying(false)}
        onImport={addTrack} onShowStem={() => setShowStem(true)} onShowEQ={() => setShowEQ(true)}
        bpm={bpm} setBpm={setBpm} />

      {/* MAIN CONTENT */}
      <div style={{ display: "flex", flex: 1, overflow: "hidden", position: "relative" }}>

        {/* TIMELINE */}
        <div style={{ flex: 1, display: "flex", flexDirection: "column", overflow: "hidden" }}>
          <TimelineRuler zoom={zoom} />

          {/* Drop Zone Banner */}
          <AnimatePresence>
            {isDragging && (
              <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
                onDragLeave={() => setIsDragging(false)}
                style={{ position: "absolute", inset: 0, zIndex: 40, background: "rgba(0,4,10,0.85)",
                  border: `2px dashed ${C.cyan}`, display: "flex", alignItems: "center", justifyContent: "center",
                  boxShadow: `inset 0 0 60px rgba(0,243,255,0.1)` }}>
                <div style={{ textAlign: "center" }}>
                  <div style={{ fontSize: 36, color: C.cyan, marginBottom: 8 }}>⊕</div>
                  <div style={{ fontFamily: "monospace", fontSize: 12, color: C.cyan, letterSpacing: 4 }}>DROP AUDIO FILES</div>
                  <div style={{ fontFamily: "monospace", fontSize: 9, color: C.textDim, letterSpacing: 3, marginTop: 4 }}>MP3, WAV, FLAC, OGG</div>
                </div>
              </motion.div>
            )}
          </AnimatePresence>

          {/* Track List */}
          <div style={{ flex: 1, overflowY: "auto", overflowX: "hidden" }}
            onDragOver={e => { e.preventDefault(); setIsDragging(true); }}
            onDrop={e => { e.preventDefault(); setIsDragging(false); addTrack(); }}>
            <AnimatePresence>
              {tracks.map(track => (
                <TrackRow key={track.id} track={track} isPlaying={isPlaying}
                  onUpdate={patch => updateTrack(track.id, patch)}
                  onDelete={() => deleteTrack(track.id)}
                  selected={selectedTrack === track.id}
                  onSelect={() => setSelectedTrack(selectedTrack === track.id ? null : track.id)} />
              ))}
            </AnimatePresence>

            {/* Add Track Button */}
            <motion.button whileHover={{ background: C.bgHover }} onClick={addTrack}
              style={{ width: "100%", padding: "10px 14px", background: "none", border: "none",
                borderBottom: `1px solid ${C.border}`, cursor: "pointer", display: "flex", alignItems: "center", gap: 8 }}>
              <span style={{ fontSize: 14, color: C.textDim }}>+</span>
              <span style={{ fontFamily: "monospace", fontSize: 9, color: C.textDim, letterSpacing: 3 }}>ADD TRACK</span>
            </motion.button>
          </div>

          <EffectsRack />
        </div>

        {/* RIGHT PANEL */}
        <motion.div initial={{ x: 240 }} animate={{ x: 0 }} transition={{ type: "spring", stiffness: 150, damping: 25 }}
          style={{ width: 220, borderLeft: `1px solid ${C.border}`, background: C.bgPanel, display: "flex", flexDirection: "column" }}>

          {/* Master */}
          <div style={{ padding: "12px 14px", borderBottom: `1px solid ${C.border}` }}>
            <div style={{ fontFamily: "monospace", fontSize: 9, color: C.cyan, letterSpacing: 3, marginBottom: 10 }}>MASTER</div>
            <div className="flex items-center gap-3">
              <div style={{ flex: 1 }}>
                <NeonSlider value={85} color={C.cyan} onChange={() => {}} />
              </div>
              <VUMeter active={isPlaying} color={C.cyan} />
            </div>
            <div style={{ marginTop: 8, display: "flex", gap: 4 }}>
              <CyberButton small active>MONO</CyberButton>
              <CyberButton small>STEREO</CyberButton>
            </div>
          </div>

          {/* Keyboard shortcuts */}
          <div style={{ padding: "12px 14px", borderBottom: `1px solid ${C.border}` }}>
            <div style={{ fontFamily: "monospace", fontSize: 9, color: C.textDim, letterSpacing: 3, marginBottom: 8 }}>SHORTCUTS</div>
            {[
              ["SPACE", "PLAY/PAUSE"],
              ["C", "CUT TOOL"],
              ["V", "SELECT"],
              ["D", "DRAW"],
              ["CTRL+Z", "UNDO"],
              ["CTRL+S", "SAVE"],
            ].map(([k, v]) => (
              <div key={k} className="flex justify-between" style={{ marginBottom: 4 }}>
                <span style={{ fontSize: 8, color: C.cyan, background: `rgba(0,243,255,0.1)`, padding: "1px 4px", letterSpacing: 1 }}>{k}</span>
                <span style={{ fontSize: 8, color: C.textDim }}>{v}</span>
              </div>
            ))}
          </div>

          {/* Stem Badges */}
          <div style={{ padding: "12px 14px" }}>
            <div style={{ fontFamily: "monospace", fontSize: 9, color: C.textDim, letterSpacing: 3, marginBottom: 8 }}>STEM STATUS</div>
            <div className="flex flex-col gap-2">
              {[["VOCALS", C.cyan], ["BASS", C.purple], ["DRUMS", C.pink], ["MELODY", C.yellow]].map(([s, c]) => (
                <div key={s} className="flex items-center justify-between">
                  <StemBadge label={s} color={c} />
                  <span style={{ fontSize: 8, color: C.textDim }}>READY</span>
                </div>
              ))}
            </div>
          </div>

          {/* Version */}
          <div style={{ marginTop: "auto", padding: "10px 14px", borderTop: `1px solid ${C.border}` }}>
            <div style={{ fontFamily: "monospace", fontSize: 7, color: C.textDim, letterSpacing: 2 }}>ALONE AUDIO STUDIO</div>
            <div style={{ fontFamily: "monospace", fontSize: 7, color: C.textDim, letterSpacing: 2 }}>v1.0.0 • CLIENT-SIDE ONLY</div>
            <div style={{ fontFamily: "monospace", fontSize: 7, color: C.cyanDim, letterSpacing: 2, marginTop: 2 }}>
              PWA • OFFLINE READY
            </div>
          </div>
        </motion.div>
      </div>

      {/* MODALS */}
      <AnimatePresence>
        {showEQ && (
          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
            style={{ position: "fixed", inset: 0, zIndex: 60, display: "flex", alignItems: "flex-end",
              justifyContent: "center", paddingBottom: 80, background: "rgba(0,0,0,0.6)" }}
            onClick={e => { if (e.target === e.currentTarget) setShowEQ(false); }}>
            <EQPanel onClose={() => setShowEQ(false)} />
          </motion.div>
        )}
        {showStem && (
          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
            style={{ position: "fixed", inset: 0, zIndex: 60, display: "flex", alignItems: "center",
              justifyContent: "center", background: "rgba(0,0,0,0.7)" }}
            onClick={e => { if (e.target === e.currentTarget) setShowStem(false); }}>
            <StemPanel onClose={() => setShowStem(false)} onProcess={handleStemDone} />
          </motion.div>
        )}
      </AnimatePresence>

      {/* Pulse animation */}
      <style>{`
        @keyframes pulse { 0%,100% { opacity:1; } 50% { opacity:0.3; } }
        ::-webkit-scrollbar { width: 6px; height: 6px; }
        ::-webkit-scrollbar-track { background: ${C.bg}; }
        ::-webkit-scrollbar-thumb { background: ${C.border}; }
        ::-webkit-scrollbar-thumb:hover { background: ${C.cyan}; }
        input[type=range]::-webkit-slider-thumb { -webkit-appearance:none; width:12px; height:12px; border-radius:2px; background:${C.cyan}; cursor:pointer; }
        input[type=number]::-webkit-inner-spin-button, input[type=number]::-webkit-outer-spin-button { opacity:0.3; }
      `}</style>
    </div>
  );
}
